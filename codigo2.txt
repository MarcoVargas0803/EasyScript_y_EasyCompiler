/* 
  Estructuras de control SEGÚN, MIENTRAS Y PARA. 
*/

INICIO{

    // PRUEBA DE SEGÚN 
    ENT Dia_semana = 3;

    SEGUN (Dia_semana) {
        CASO 1: IMPRIMIR("Lunes"); DETENER;
        CASO 2: IMPRIMIR("Martes"); DETENER;
        CASO 3: IMPRIMIR("Miercoles"); DETENER;
        CASO 4: IMPRIMIR("Jueves"); DETENER;
        CASO 5: IMPRIMIR("Viernes"); DETENER;
        DEFECTO: IMPRIMIR("Fin de semana o error"); 
    }   
    
    // PRUEBA MIENTRAS (CONTADOR)
    ENT Contador = 1;

    MIENTRAS (Contador <= 5) {
        IMPRIMIR("Ciclo Mientras: " + Contador);
        Contador = Contador + 1;
    }

    //Prueba PARA 
    PARA (ENT Cont = 1; Cont <= 5; Cont--) {
        IMPRIMIR("Ciclo Para: " + Cont);
    }

}FIN
